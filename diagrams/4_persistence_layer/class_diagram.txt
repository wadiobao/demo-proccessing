@startuml
title Layer: Data Persistence

package "JPA / SQL (PostgreSQL)" {
    class User {
        <<@Entity>>
        -Long id
        -String userName
        -String password
        -String email
        -Set<String> roles
    }
    class Topic {
        <<@Entity>>
        -Long id
        -String name
    }
    interface UserRepository {
        +findByUserName(String) : Optional<User>
    }
    interface TopicRepository {
    }
    
    UserRepository ..> User
    TopicRepository ..> Topic
}

package "Mongo (NoSQL)" {
    class PdfStore {
        <<@Document>>
        -String id
        -String author
        -String content
        -String wordBase64
    }
    class QuestionBank {
        <<@Document>>
        -String id
        -String contentId
        -String questionHash
        -Question questionData
        -int attempts
        -int correctCount
        -double difficulty
    }
    interface PdfStoreRepository {
        +save(PdfStore) : PdfStore
    }
    interface QuestionBankRepository {
        <<Mongo>>
        +countByContentId(String) : long
        +save(QuestionBank) : QuestionBank
        +findByContentIdAndQuestionHash(String, String) : Optional<QuestionBank>
    }
    
    PdfStoreRepository ..> PdfStore
    QuestionBankRepository ..> QuestionBank
}

@enduml
