@startuml
title Layer: Data Persistence

package "JPA / SQL (PostgreSQL)" {
    class User {
        <<@Entity>>
        -Long id
        -String userName
        -String password
        -String email
        -Set<String> roles
    }
    class Topic {
        <<@Entity>>
        -Long id
        -String name
    }
    interface UserRepository {
        +findByUserName(String) : Optional<User>
    }
    interface TopicRepository {
    }
    
    UserRepository ..> User
    TopicRepository ..> Topic
}

package "Mongo (NoSQL)" {
    class PdfStore {
        <<@Document>>
        -String id
        -String author
        -String content
        -String wordBase64
    }
    interface PdfStoreRepository {
        +save(PdfStore) : PdfStore
    }
    
    PdfStoreRepository ..> PdfStore
}

@enduml
