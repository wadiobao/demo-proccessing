@startuml
title Activity Diagram - AI Quiz Generation from PDF (Refactored)

|Client|
start
:Tải lên file PDF và QuizConfig;

|QuizController|
:Nhận request;
:Gọi QuizService.processPrivateQuiz();

|QuizService|
:Gọi QuizProcessor.processQuiz();

|QuizProcessor|
:PDFProcessingService.extractTextFromPdf();
:Xây dựng Prompt với PromptBuilder;
:Gọi GeminiAIUtils.generateQuestionWithGemini();
:Phân tích phản hồi với ResponseBuilder;
:Gọi WordPdfGeneration.generateWordAndPdfBase64();
:Tạo StateResponse;

|QuizService|
:Lấy kết quả từ QuizProcessor;
:Lấy Authentication từ SecurityContextHolder;
if (Người dùng đã đăng nhập?) then (yes)
    |QuizPersistenceManager|
    :Lưu dữ liệu vào MongoDB;
    :Cập nhật Response với metadata;
endif

|QuizController|
:Trả về StateResponse;

|Client|
:Nhận kết quả và link download;
stop

@enduml
