@startuml
title Activity Diagram - User Login

|Client|
start
:Gửi yêu cầu đăng nhập (username, password);

|AuthenticationController|
:Nhận AuthenticationUser DTO;
:Gọi AuthenticationService.authenticate();

|AuthenticationService|
:Tìm người dùng trong DB qua UserRepository;
|UserRepository|
:findByUserName(username);

|AuthenticationService|
if (Tìm thấy user?) then (yes)
    :Sử dụng BCryptPasswordEncoder;
    :So khớp password nhập vào với password trong DB;
    if (Khớp?) then (yes)
        :Tạo Access Token và Refresh Token (JWT);
        :Tạo Access Cookie và Refresh Cookie;
        :Trả về ResponseEntity chứa kết quả;
    else (no)
        :Ném HandleException(UNAUTHENTICATED);
    endif
else (no)
    :Ném HandleException;
endif

|AuthenticationController|
:Nhận kết quả;
:Trả về HTTP 200 OK kèm Cookies;

|Client|
:Nhận Cookies;
:Chuyển hướng người dùng;
stop

@enduml
