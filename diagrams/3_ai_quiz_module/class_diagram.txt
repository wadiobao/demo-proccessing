@startuml
title Module: AI-Powered Quiz Generation

package "Controller" {
    class QuizController {
        -IQuizService quizService
        +handlePdf(MultipartFile, QuizConfig) : StateResponse~Object~
    }
}

package "Service" {
    class QuizService {
        -QuizProcessor quizProcessor
        -QuizPersistenceManager persistenceManager
        -QuizAnswerService quizAnswerService
        +processPublicQuiz(MultipartFile, QuizConfig) : StateResponse~Object~
        +processPrivateQuiz(MultipartFile, QuizConfig) : StateResponse~Object~
    }

    class QuizAnswerService {
        -UserResourceRepository userResourceRepository
        -IRTCalculator irtCalculator
        -QuestionBankRepository questionBankRepository
        +submitQuizAnswers(QuizSubmissionRequest, String) : StateResponse~Object~
    }

    class QuizProcessor {
        -PDFProcessingService pdfProcessingService
        -GeminiAIUtils geminiAIUtils
        -QuizPromptBuilder promptBuilder
        -QuizResponseBuilder responseBuilder
        -WordPdfGeneration wordPdfGeneration
        -QuestionBankRepository questionBankRepository
        +processQuiz(MultipartFile, QuizConfig, String contentId) : StateResponse~Object~
    }

    class QuizPersistenceManager {
        -PdfStoreRepository pdfStoreRepo
        -UserResourceService userResourceService
        -QuestionBankRepository questionBankRepository
        -GeneralUtils generalUtils
        +persistQuizData(FileGenerateResponse, String, String, String contentId) : FileGenerateResponse
    }

    class PDFProcessingService {
        +extractTextFromPdf(MultipartFile) : String
    }

    class GeminiAIUtils {
        +generateQuestionWithGemini(String) : String
    }
    
    class WordPdfGeneration {
        +generateWordAndPdfBase64(List~Question~) : String[]
    }
}

package "Repository" {
    interface PdfStoreRepository {
        <<Mongo>>
        +save(PdfStore) : PdfStore
    }
    interface QuestionBankRepository {
        <<Mongo>>
        +countByContentId(String) : long
        +save(QuestionBank) : QuestionBank
        +findByContentIdAndQuestionHash(String, String) : Optional~QuestionBank~
    }
    interface UserResourceRepository {
        <<Mongo>>
        +findByUserNameAndTopic(String, String) : Optional~UserResource~
    }
}

package "Model" {
    class PdfStore {
        -String id
        -String author
        -String content
        -String wordBase64
    }
    class QuestionBank {
        -String id
        -String contentId
        -String questionHash
        -Question questionData
        -int attempts
        -int correctCount
        -double difficulty
    }
    class Question {
        -String id
        -String bankId
        -String question
        -List~Answer~ answers
    }
}

QuizController o-- QuizService
QuizService o-- QuizProcessor
QuizService o-- QuizPersistenceManager
QuizService o-- QuizAnswerService
QuizAnswerService o-- IRTCalculator
QuizAnswerService o-- UserResourceRepository
QuizAnswerService o-- QuestionBankRepository
QuizProcessor o-- QuestionBankRepository
QuizPersistenceManager o-- QuestionBankRepository
QuestionBankRepository ..> QuestionBank
QuestionBank o-- Question
QuizProcessor ..> Question

@enduml
