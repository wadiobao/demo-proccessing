@startuml
title Module: Authentication & User Management

package "Controller" {
    class AuthenticationController {
        -AuthenticationService authService
        +authenticate(AuthenticationUser) : ResponseEntity
        +introspect(Introspect) : IntrospectResponse
        +refreshToken(HttpServletRequest, String) : ResponseEntity
        +logout(HttpServletResponse, String) : void
    }

    class UserController {
        -UserService userService
        +getAll(Pageable) : Page~User~
        +registerUser(UserRequest) : UserResponse
        +getInfor(String) : UserResponse
        +myInfor() : UserResponse
    }
}

package "Service" {
    class AuthenticationService {
        -UserRepository repository
        -InvalidatedTokenRepository invalidatedTokenRepository
        -JwtUtils jwtUtils
        +authenticate(AuthenticationUser) : ResponseEntity
        +introspect(Introspect) : IntrospectResponse
        +refreshToken(HttpServletResponse, String) : ResponseEntity
        +logout(HttpServletResponse, String) : void
    }

    class UserService {
        -UserRepository userRepository
        -PasswordEncoder encoder
        +getAll(Pageable) : Page~User~
        +registerUser(UserRequest) : UserResponse
        +getInfor(String) : UserResponse
        +myInfor() : UserResponse
    }
}

package "Repository" {
    interface UserRepository {
        +findByUserName(String) : Optional~User~
        +existsByUserName(String) : boolean
        +existsByEmail(String) : boolean
    }
    interface InvalidatedTokenRepository {
        +save(InvalidatedToken) : InvalidatedToken
    }
}

package "Model/Entity" {
    class User {
        -Long id
        -String userName
        -String password
        -String email
        -Set~String~ roles
    }
    class InvalidatedToken {
        -String id
        -Date expiryTime
    }
}

package "Utils" {
    class JwtUtils {
        +generateToken(User, boolean) : String
        +verifyToken(String) : SignedJWT
        +generateAccessCookie(String) : ResponseCookie
        +generateRefreshCookie(String) : ResponseCookie
    }
}

AuthenticationController o-- AuthenticationService
UserController o-- UserService
AuthenticationService o-- UserRepository
AuthenticationService o-- InvalidatedTokenRepository
AuthenticationService o-- JwtUtils
UserService o-- UserRepository
UserRepository ..> User
InvalidatedTokenRepository ..> InvalidatedToken

@enduml
